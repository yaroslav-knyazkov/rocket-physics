\begin{gather}
    \vec{W} = mg\hat{z}_g \\
    \vec{T} = T\hat{x}_r \\
    \vec{A}_f = -A_x\hat{x}_r - A_z\hat{z}_r
\end{gather}
\subsection{Moments}
\begin{equation}
    \vec{A}_m = -A_m\hat{y}_r
\end{equation}
\subsection{Geometry}
Measured
\begin{gather}
    \vec{Aa} = -l_1\hat{x}_r \\
    \vec{AB} = -l_2\hat{x}_r
\end{gather}
Calculated
\begin{gather}
    \vec{aA} = l_1\hat{x}_r \\
    \vec{aB} = (l_1 - l_2)\hat{x}_r
\end{gather}
\subsection{Equations of motion}
\begin{gather}
    m\vec{\ddot{x}} = \sum \vec{F} = \vec{W} + \vec{T} + \vec{A}_f \\
    I\ddot{\theta} = \sum \vec{M}(a) \cdot \hat{y} = \left[\vec{\vec{aB}}\times\vec{T} + \vec{aA}\times\vec{A}_f + \vec{A_m}\right]\cdot \hat{y}
\end{gather}
